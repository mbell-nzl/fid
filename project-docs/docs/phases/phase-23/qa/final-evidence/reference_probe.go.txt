package workflow

import (
 "encoding/json"
 "testing"
)

func TestQAPhase23ReferenceAttentionKeyAndBytes(t *testing.T) {
 t.Parallel()
 name := "attempts"
 capability := "attention.v1"
 for _, kind := range []string{"executor-terminated","run-recovered","activation-recovered","capability-available","budget-observed","work-ended"} {
  subject := Subject{Type:"run", ID:"RUN-pinned"}
  if kind == "activation-recovered" || kind == "capability-available" { subject = Subject{Type:"activation", ID:"ACT-pinned"} }
  condition := Condition{Type:kind, Subject:subject}
  if kind == "budget-observed" { condition.BudgetName = &name }
  if kind == "capability-available" { condition.CapabilityID = &capability }
  raw,err := json.Marshal(condition); if err != nil {t.Fatal(err)}
  t.Logf("canonical %s: %s",kind,raw)
 }
 subject := Subject{Type:"run", ID:"RUN-pinned"}
 condition := Condition{Type:"budget-observed",Subject:subject,BudgetName:&name}
 key,err := attentionSemanticKey(subject,"budget-exhausted",condition)
 if err != nil {t.Fatal(err)}
 t.Logf("subject={type:run,id:RUN-pinned} code=budget-exhausted condition={type:budget-observed,subject:{type:run,id:RUN-pinned},budget_name:attempts,capability_id:null} key=%s",key)
 const want="sha256:d4b51bb4a6404d4a2579826ad5c08d802ed83c41814cd5fa74d3ee8c7796c3de"
 if key != want {t.Fatalf("key=%s want=%s",key,want)}
}
